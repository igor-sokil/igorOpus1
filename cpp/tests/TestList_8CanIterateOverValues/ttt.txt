TEST_CASE(SUITE("8CanIterateOverValues"))
{
    List<int> list(3);

    list.Add(1);
    list.Add(2);
    list.Add(3);

    auto iter = list.Iterate();

    for (int i = 1; i < 4; ++i)
    {
        REQUIRE(iter.HasNext());
        REQUIRE(i == iter.Next()->value);
    }

    REQUIRE(!iter.HasNext());
}
